// Title: Implementation of Basic JavaScript Techniques for a Smart City Information Portal // Aim: To implement basic JavaScript concepts such as // variables, data types, operators, input/output methods, and conditional statements to add interactivity to a Municipal Corporation Smart City Information Portal. // 1. Accepts the citizen's name and age var citizenName = prompt("Enter citizen name:"); var citizenAge = parseInt(prompt("Enter citizen age:")); // 2. Displays a personalized welcome message to the citizen alert("Welcome " + citizenName + " to the Smart City Information Portal!"); // 3. Accepts the citizen's ward number var wardNumber = parseInt(prompt("Enter your ward number:")); // 4. Checks whether the entered ward number is valid. Assume valid ward numbers are 1 to 50. var wardStatus = ""; if (wardNumber >= 1 && wardNumber <= 50) { wardStatus = "Ward " + wardNumber + " is a valid ward number."; } else { wardStatus = "Ward " + wardNumber + " is an invalid ward number. (Valid ward numbers are 1 to 50)"; } alert(wardStatus); // 5. Determines whether the citizen is eligible for a Senior Citizen Service based on age: // Age 60 or above -> Eligible // Below 60 -> Not Eligible var seniorCitizenStatus = ""; if (citizenAge >= 60) { seniorCitizenStatus = "Eligible for Senior Citizen Service."; } else { seniorCitizenStatus = "Not Eligible for Senior Citizen Service."; } alert(seniorCitizenStatus); // 6. Accepts two monthly water-meter readings and calculates the number of units consumed var previousReading = parseFloat(prompt("Enter previous month water-meter reading:")); var currentReading = parseFloat(prompt("Enter current month water-meter reading:")); var unitsConsumed = currentReading - previousReading; // 7. Displays the calculated water consumption alert("Calculated Water Consumption: " + unitsConsumed + " units"); // 8. Displays appropriate messages based on the entered information var summaryMessage = "Citizen Information Summary:\n\n" + "Name: " + citizenName + "\n" + "Age: " + citizenAge + "\n" + "Ward Number: " + wardNumber + " (" + wardStatus + ")\n" + "Senior Citizen Service: " + seniorCitizenStatus + "\n" + "Previous Water Reading: " + previousReading + "\n" + "Current Water Reading: " + currentReading + "\n" + "Total Units Consumed: " + unitsConsumed + " units"; alert(summaryMessage);
